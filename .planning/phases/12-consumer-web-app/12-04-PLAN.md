```tsx
import { MapContainer, TileLayer, Marker, Popup } from 'react-leaflet';
import L from 'leaflet';
// CRITICAL: import 'leaflet/dist/leaflet.css'; in main.tsx (not in component)
// Default marker icon fix (Leaflet + Vite issue):
import markerIcon2x from 'leaflet/dist/images/marker-icon-2x.png';
import markerIcon from 'leaflet/dist/images/marker-icon.png';
import markerShadow from 'leaflet/dist/images/marker-shadow.png';
L.Icon.Default.mergeOptions({ iconRetinaUrl: markerIcon2x, iconUrl: markerIcon, shadowUrl: markerShadow });

// MapContainer usage:
<MapContainer center={[lat, lng]} zoom={13} style={{ height: '70vh', width: '100%' }}>
  <TileLayer
    url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
    attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>'
  />
  {stores.map((store) => (
    <Marker key={store.id} position={[store.location!.coordinates[1], store.location!.coordinates[0]]}>
      <Popup>
        <strong>{store.name}</strong><br />
        {store.address}
      </Popup>
    </Marker>
  ))}
</MapContainer>
// IMPORTANT: coordinates from PostGIS are [lng, lat] order — Leaflet expects [lat, lng]
// Extract: lat = store.location.coordinates[1], lng = store.location.coordinates[0]
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser geolocation → store query | User location sent to GET /stores/ endpoint |
| Leaflet tile requests → OpenStreetMap CDN | External tile requests; no auth credentials involved |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-10 | Information Disclosure | User geolocation in getNearby request | accept | User explicitly grants browser geolocation permission; fallback to city center avoids location disclosure for users who deny; coordinates not persisted client-side |
| T-12-11 | Spoofing | Leaflet Popup links | mitigate | Popup links use hardcoded /app/map/store/ path pattern; store.id is an opaque string from the API — no script injection surface since React/Leaflet render via DOM, not innerHTML |
| T-12-12 | Denial of Service | Map loading many markers | accept | Marker count bounded by the server-side radius filter (10 km default); no client-side loop risk |
</threat_model>
