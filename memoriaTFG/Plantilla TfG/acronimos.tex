% Glosario de acrónimos
\chapter*{Acrónimos}
\addcontentsline{toc}{chapter}{Acrónimos}

\begin{description}
  \item[ADR] \emph{Architecture Decision Record}: registro de decisión arquitectónica.
  \item[API] \emph{Application Programming Interface}: interfaz de programación de aplicaciones.
  \item[CI/CD] \emph{Continuous Integration / Continuous Delivery}: integración y entrega continuas.
  \item[CRUD] \emph{Create, Read, Update, Delete}: operaciones básicas de persistencia.
  \item[CSV] \emph{Comma-Separated Values}: formato tabular de texto plano.
  \item[DRF] \emph{Django REST Framework}: framework para construir APIs REST sobre Django.
  \item[E2E] \emph{End-to-End}: pruebas de extremo a extremo del sistema integrado.
  \item[EAN] \emph{European Article Number}: código de barras normalizado de producto.
  \item[GIN] \emph{Generalized Inverted Index}: índice invertido generalizado de PostgreSQL.
  \item[GiST] \emph{Generalized Search Tree}: familia de índices de PostgreSQL usada por PostGIS.
  \item[HMR] \emph{Hot Module Replacement}: recarga en caliente de módulos en desarrollo.
  \item[HU] Historia de Usuario.
  \item[IA] Inteligencia Artificial.
  \item[JSON] \emph{JavaScript Object Notation}: formato de intercambio de datos.
  \item[JWT] \emph{JSON Web Token}: estándar de testigos de autenticación firmados.
  \item[LLM] \emph{Large Language Model}: modelo de lenguaje de gran tamaño.
  \item[LSSI-CE] Ley de Servicios de la Sociedad de la Información y de Comercio Electrónico.
  \item[OCR] \emph{Optical Character Recognition}: reconocimiento óptico de caracteres.
  \item[ORM] \emph{Object-Relational Mapping}: mapeo objeto-relacional.
  \item[ORS] OpenRouteService: servicio de cálculo de rutas sobre OpenStreetMap.
  \item[PYME] Pequeña y Mediana Empresa.
  \item[REST] \emph{Representational State Transfer}: estilo arquitectónico de APIs web.
  \item[RF] Requisito Funcional.
  \item[RGPD] Reglamento General de Protección de Datos (Reglamento (UE) 2016/679).
  \item[RI] Requisito de Información.
  \item[RN] Regla de Negocio.
  \item[RNF] Requisito No Funcional.
  \item[SDK] \emph{Software Development Kit}: kit de desarrollo de software.
  \item[SPA] \emph{Single-Page Application}: aplicación web de página única.
  \item[SRID] \emph{Spatial Reference System Identifier}: identificador de sistema de referencia espacial.
  \item[TFG] Trabajo Fin de Grado.
  \item[TSP] \emph{Travelling Salesman Problem}: problema del viajante de comercio.
  \item[TTL] \emph{Time To Live}: tiempo de vigencia de un dato.
  \item[UAT] \emph{User Acceptance Testing}: pruebas de aceptación de usuario.
  \item[UI] \emph{User Interface}: interfaz de usuario.
  \item[URL] \emph{Uniform Resource Locator}: localizador uniforme de recursos.
  \item[UX] \emph{User Experience}: experiencia de usuario.
  \item[VRP] \emph{Vehicle Routing Problem}: problema de enrutamiento de vehículos.
  \item[WCAG] \emph{Web Content Accessibility Guidelines}: pautas de accesibilidad para contenido web.
\end{description}
